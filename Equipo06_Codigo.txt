// Ayudas compartidas por los specs (no es un spec: no termina en .cy.js).
// Todo Supabase se simula con cy.intercept; nunca se toca la base real.

// Mismo formato que usa la app para los precios
export const pesos = (valor) => `$ ${Number(valor).toLocaleString('es-AR')}`

// Usuario y perfil de prueba (datos inventados)
export const USUARIO = { id: '00000000-0000-4000-8000-000000000001', email: 'cliente@prueba.test' }
export const PERFIL = { nombre: 'Ana Prueba', rol: 'cliente' }

const CORS = {
  'access-control-allow-origin': '*',
  'access-control-allow-headers':
    'authorization, apikey, content-type, x-client-info, prefer, accept, accept-profile, content-profile',
  'access-control-allow-methods': 'GET, POST, OPTIONS',
  'content-type': 'application/json',
}

// Responde con JSON (acepta null para simular una respuesta vacía)
export function responder(req, cuerpo, estado = 200) {
  req.reply({ statusCode: estado, headers: CORS, body: JSON.stringify(cuerpo) })
}

// Fábrica de productos de ejemplo
let contador = 0
export function producto(datos = {}) {
  contador += 1
  return {
    id: `10000000-0000-4000-8000-${String(contador).padStart(12, '0')}`,
    nombre: `Producto ${contador}`,
    descripcion: 'Descripción de prueba',
    info_nutricional: null,
    precio: 1000,
    stock: 5,
    unidad_venta: 'unidad',
    sin_tacc: false,
    vegano: false,
    sin_lactosa: false,
    categoria_id: null,
    imagen_path: null,
    publicado: true,
    ...datos,
  }
}

// Token con forma de JWT (solo para que supabase-js lo acepte; no vale nada)
function tokenFalso() {
  const base64url = (obj) =>
    btoa(JSON.stringify(obj)).replace(/=+$/, '').replace(/\+/g, '-').replace(/\//g, '_')
  const payload = {
    sub: USUARIO.id,
    aud: 'authenticated',
    role: 'authenticated',
    email: USUARIO.email,
    exp: Math.floor(Date.now() / 1000) + 3600,
  }
  return `${base64url({ alg: 'HS256', typ: 'JWT' })}.${base64url(payload)}.firma-falsa`
}

export function usuarioAuth(extra = {}) {
  return {
    id: USUARIO.id,
    aud: 'authenticated',
    role: 'authenticated',
    email: USUARIO.email,
    app_metadata: {},
    user_metadata: { nombre: PERFIL.nombre },
    identities: [{ id: USUARIO.id, provider: 'email' }],
    created_at: '2026-01-01T00:00:00Z',
    ...extra,
  }
}

// Debe llamarse primero: lo que no tenga un intercept propio responde 500 y queda
// contado en @noPrevisto (ningún request llega a la base real).
export function bloquearSupabase() {
  cy.intercept(
    { url: /\.supabase\.co|\/(rest|auth|functions|storage)\/v1\// },
    (req) => responder(req, { message: 'Request de Supabase no previsto en el test' }, 500)
  ).as('noPrevisto')
}

// Para usar en afterEach: confirma que no se coló ningún request sin simular
export function verificarSinNoPrevistos() {
  cy.get('@noPrevisto.all').should('have.length', 0)
}

// Catálogo: lista de productos (o uno solo si piden por id) y categorías
export function simularCatalogo(productos, categorias = []) {
  cy.intercept({ method: 'GET', pathname: '/rest/v1/categorias' }, (req) =>
    responder(req, categorias)
  )
  cy.intercept({ method: 'GET', pathname: '/rest/v1/productos' }, (req) => {
    const id = new URL(req.url).searchParams.get('id')
    if (id) {
      responder(req, productos.find((p) => `eq.${p.id}` === id) ?? null)
    } else {
      responder(req, productos)
    }
  }).as('productos')
}

// Zonas de cobertura: costos por código postal; los demás códigos no tienen zona (null)
export function simularZonas(costos = { 1754: 1500 }) {
  cy.intercept({ method: 'GET', pathname: '/rest/v1/zonas_cobertura' }, (req) => {
    const cp = (new URL(req.url).searchParams.get('codigo_postal') ?? '').replace('eq.', '')
    responder(req, cp in costos ? { costo_envio: costos[cp] } : null)
  })
}

// Login simulado: token de contraseña y perfil
export function simularSesion() {
  cy.intercept({ method: 'POST', pathname: '/auth/v1/token' }, (req) =>
    responder(req, {
      access_token: tokenFalso(),
      token_type: 'bearer',
      expires_in: 3600,
      refresh_token: 'refresh-falso',
      user: usuarioAuth(),
    })
  )
  cy.intercept({ method: 'GET', pathname: '/rest/v1/perfiles' }, (req) => responder(req, PERFIL))
}

// Prepara todo lo necesario para una pantalla con catálogo y sesión posible
export function prepararTienda({ productos, categorias = [] }) {
  bloquearSupabase()
  simularCatalogo(productos, categorias)
  simularSesion()
}

// Entra por la pantalla de ingreso (sin salir del router, para no perder el carrito).
// La contraseña es aleatoria: el intercept acepta cualquiera.
export function ingresarPorUI() {
  cy.contains('a', 'Cuenta').click()
  cy.get('#correo').type(USUARIO.email)
  cy.get('#contrasenia').type(`x${Cypress._.random(1e9)}`, { log: false })
  cy.contains('button', 'Entrar').click()
  cy.contains('button', 'Salir').should('be.visible')
}

// Primera unidad: "Agregar". Si ya está en el carrito, el + de la tarjeta.
export function agregarDesdeCatalogo(nombre) {
  cy.contains('.tarjeta', nombre).then(($tarjeta) => {
    const sumar = $tarjeta.find(`button[aria-label="Sumar una unidad de ${nombre}"]`)
    if (sumar.length) {
      cy.wrap(sumar).click()
    } else {
      cy.wrap($tarjeta).contains('button', 'Agregar').click()
    }
  })
}

// El contador va en la insignia, no en el texto "Carrito (n)".
export function verCarrito(cantidad) {
  cy.get('a.header-carrito').find('.cuenta').should('have.text', String(cantidad))
}

// Va al carrito con el link del Header (no recarga la página)
export function irAlCarrito() {
  cy.get('a.header-carrito').click()
}

describe('Carrito', () => {
  afterEach(verificarSinNoPrevistos)

  it('HU-04: agregar, sumar con tope de stock, restar, quitar y vaciar', () => {
    const harina = producto({ nombre: 'Harina de arroz', precio: 1200, stock: 3 })
    const miel = producto({ nombre: 'Miel de campo', precio: 2500, stock: 5 })
    prepararTienda({ productos: [harina, miel] })

    const subtotal = (total) => cy.get('.carrito-cierre strong').should('have.text', pesos(total))
    const fila = (nombre) => cy.contains('.carrito-fila', nombre)

    cy.visit('/')

    // El mismo producto dos veces: una fila con cantidad 2 y Header (2)
    agregarDesdeCatalogo('Harina de arroz')
    agregarDesdeCatalogo('Harina de arroz')
    verCarrito(2)
    agregarDesdeCatalogo('Miel de campo')
    irAlCarrito()
    cy.get('.carrito-fila').should('have.length', 2)
    fila('Harina de arroz').find('.cantidad span').should('have.text', '2')
    subtotal(1200 * 2 + 2500)

    // El + llega al tope de stock (3) y se deshabilita
    fila('Harina de arroz').find('.cantidad button').last().click()
    fila('Harina de arroz').find('.cantidad span').should('have.text', '3')
    fila('Harina de arroz').find('.cantidad button').last().should('be.disabled')
    subtotal(1200 * 3 + 2500)

    // El − hasta 0 quita la fila
    for (let i = 0; i < 3; i++) {
      fila('Harina de arroz').find('.cantidad button').first().click()
    }
    cy.contains('.carrito-fila', 'Harina de arroz').should('not.exist')
    subtotal(2500)

    // Sin sesión, seguir comprando pide ingreso y vuelve al catálogo
    cy.contains('a', 'Seguir comprando').click()
    cy.location('pathname').should('eq', '/ingreso')
    ingresarPorUI()
    agregarDesdeCatalogo('Harina de arroz')
    irAlCarrito()
    subtotal(1200 + 2500)
    fila('Harina de arroz').contains('button', 'Quitar').click()
    cy.contains('.carrito-fila', 'Harina de arroz').should('not.exist')
    subtotal(2500)

    // "Vaciar carrito" deja el carrito vacío
    cy.contains('button', 'Vaciar carrito').click()
    cy.contains('Tu carrito está vacío').should('be.visible')
    verCarrito(0)
  })

  it('HU-04/HU-05: "Ir a pagar" solo con sesión y con productos', () => {
    const harina = producto({ nombre: 'Harina de arroz', precio: 1200, stock: 3 })
    prepararTienda({ productos: [harina] })

    cy.visit('/')

    // Sin sesión y con producto: no hay botón
    agregarDesdeCatalogo('Harina de arroz')
    irAlCarrito()
    cy.contains('.carrito-fila', 'Harina de arroz').should('exist')
    cy.contains('button', 'Ir a pagar').should('not.exist')

    // Con sesión y producto: aparece (el carrito se conserva al ingresar)
    ingresarPorUI()
    irAlCarrito()
    cy.contains('button', 'Ir a pagar').should('be.visible')

    // Con sesión y carrito vacío: no hay botón
    cy.contains('button', 'Vaciar carrito').click()
    cy.contains('Tu carrito está vacío').should('be.visible')
    cy.contains('button', 'Ir a pagar').should('not.exist')

    // Con sesión y un producto: el botón lleva a /pago
    cy.contains('a', 'Ver productos').click()
    agregarDesdeCatalogo('Harina de arroz')
    irAlCarrito()
    cy.contains('button', 'Ir a pagar').click()
    cy.location('pathname').should('eq', '/pago')
  })
})

import { prepararTienda, producto, verCarrito, verificarSinNoPrevistos } from './ayudas'

describe('Catálogo', () => {
  afterEach(verificarSinNoPrevistos)

  it('HU-03: los filtros se combinan sin recargar', () => {
    const almacen = { id: 'cat-almacen', nombre: 'Almacén' }
    const bebidas = { id: 'cat-bebidas', nombre: 'Bebidas' }
    const harina = producto({ nombre: 'Harina de arroz', sin_tacc: true, vegano: true, categoria_id: almacen.id })
    const galletas = producto({ nombre: 'Galletas de avena', sin_tacc: true, vegano: true, sin_lactosa: true, categoria_id: bebidas.id })
    const leche = producto({ nombre: 'Leche de almendras', vegano: true, sin_lactosa: true, categoria_id: almacen.id })
    const pan = producto({ nombre: 'Pan de molde', sin_tacc: true, categoria_id: bebidas.id })
    const sal = producto({ nombre: 'Sal marina', categoria_id: almacen.id }) // sin etiquetas

    prepararTienda({ productos: [harina, galletas, leche, pan, sal], categorias: [almacen, bebidas] })
    cy.visit('/')
    cy.get('.tarjeta').should('have.length', 5)

    // En desarrollo React ejecuta el efecto dos veces (StrictMode), así que no se
    // compara contra 1: se anota cuántos GET hubo al cargar y se compara al final
    let pedidosAlCargar
    cy.get('@productos.all').then((pedidos) => {
      pedidosAlCargar = pedidos.length
    })

    const filtro = (texto) => cy.contains('label', texto).find('input')

    // Sin TACC + Vegano: solo los que tienen ambas etiquetas
    filtro('Sin TACC').check()
    filtro('Vegano').check()
    cy.get('.tarjeta').should('have.length', 2)
    cy.contains('.tarjeta', 'Harina de arroz').should('exist')
    cy.contains('.tarjeta', 'Galletas de avena').should('exist')
    // El producto sin etiquetas desaparece con filtros activos
    cy.contains('.tarjeta', 'Sal marina').should('not.exist')

    // Sumar una categoría restringe más
    cy.contains('button.chip', 'Almacén').click()
    cy.get('.tarjeta').should('have.length', 1)
    cy.contains('.tarjeta', 'Harina de arroz').should('exist')

    // Volver a "Todas" devuelve productos
    cy.contains('button.chip', 'Todas').click()
    cy.get('.tarjeta').should('have.length', 2)

    // Desactivar un filtro también devuelve productos
    filtro('Vegano').uncheck()
    cy.get('.tarjeta').should('have.length', 3)
    cy.contains('.tarjeta', 'Pan de molde').should('exist')
    cy.contains('.tarjeta', 'Sal marina').should('not.exist')

    // Sin filtros vuelve el producto sin etiquetas
    filtro('Sin TACC').uncheck()
    cy.get('.tarjeta').should('have.length', 5)

    // Todo se filtró en el navegador: los filtros no volvieron a pedir productos
    cy.get('@productos.all').then((pedidos) => {
      expect(pedidos.length).to.equal(pedidosAlCargar)
    })
  })

  it('HU-02: un producto sin stock no se puede agregar y uno con stock sí', () => {
    const agotado = producto({ nombre: 'Avena agotada', stock: 0 })
    const disponible = producto({ nombre: 'Miel de campo', stock: 4 })

    prepararTienda({ productos: [agotado, disponible] })
    cy.visit('/')

    // Catálogo: se ve, dice sin stock y no tiene botón Agregar
    cy.contains('.tarjeta', 'Avena agotada').within(() => {
      cy.contains('Sin stock').should('be.visible')
      cy.contains('button', 'Agregar').should('not.exist')
    })

    // Ficha: "No disponible" y sin "Agregar al carrito"
    cy.contains('h2 a', 'Avena agotada').click()
    cy.contains('.disponible', 'No disponible').should('be.visible')
    cy.contains('button', 'Agregar al carrito').should('not.exist')
    cy.contains('a', 'Volver al catálogo').click()

    // Con stock: se agrega y el Header lo refleja
    verCarrito(0)
    cy.contains('.tarjeta', 'Miel de campo').contains('button', 'Agregar').click()
    verCarrito(1)
  })
})

  USUARIO,
  agregarDesdeCatalogo,
  ingresarPorUI,
  irAlCarrito,

  producto,

const harina = producto({ nombre: 'Harina de arroz', precio: 1200, stock: 5 })

// Pedido que devolvería la función cuando todo sale bien
const pedidoOk = (cantidad) => ({
  id: 4821,
  estado: 'en_preparacion',
  items: [{ productoId: harina.id, nombre: harina.nombre, cantidad, precioUnitario: harina.precio }],
  total: harina.precio * cantidad,
  modalidad: 'retiro',
  mail_ok: true,
})

// Entra, agrega la harina `cantidad` veces y llega a /pago con clics (el carrito vive en memoria)
function llegarAPago(cantidad = 2) {
  cy.visit('/')
  ingresarPorUI()
  for (let i = 0; i < cantidad; i++) agregarDesdeCatalogo(harina.nombre)
  irAlCarrito()
  cy.contains('button', 'Ir a pagar').click()
  cy.location('pathname').should('eq', '/pago')
}

// Paso 1 con retiro en el local y datos de contacto.
// Nombre y correo vienen precargados y la pantalla vuelve a llenarlos si quedan vacíos,
// por eso se pisan con {selectall} en vez de usar clear() (que deja el correo duplicado).
function completarPaso1Retiro() {
  cy.contains('label', 'Retiro en el local').find('input').check()
  cy.get('#nombre-pedido').type('{selectall}Ana Prueba')
  cy.get('#telefono').type('1100000000')
  cy.get('#correo-pedido').type(`{selectall}${USUARIO.email}`)
  cy.contains('button', 'Continuar').click()
  cy.contains('Cobro simulado').should('be.visible')
}

function elegirResultado(texto) {
  cy.contains('label', texto).find('input').check()
  cy.contains('button', 'Confirmar pedido').click()
}

describe('Checkout', () => {
  beforeEach(() => {
    prepararTienda({ productos: [harina] })
    simularZonas({ 1754: 1500 })
  })

  afterEach(verificarSinNoPrevistos)

  it('HU-08: pago aprobado confirma el pedido y vacía el carrito', () => {
    cy.intercept({ method: 'POST', pathname: '/functions/v1/confirmar-pedido' }, (req) =>
      responder(req, pedidoOk(2), 201)
    ).as('confirmar')

    llegarAPago(2)
    completarPaso1Retiro()
    elegirResultado('Aprobado')

    // El body trae solo ids y cantidades: sin precios ni datos de tarjeta
    cy.wait('@confirmar').then(({ request }) => {
      const body = typeof request.body === 'string' ? JSON.parse(request.body) : request.body
      expect(body.resultadoCobro).to.eq('aprobado')
      expect(body.items).to.deep.eq([{ productoId: harina.id, cantidad: 2 }])
      expect(JSON.stringify(body)).not.to.match(/precio|tarjeta|card|cvv|vencimiento/i)
    })

    // Pantalla de confirmación
    cy.location('pathname').should('eq', '/confirmacion')
    cy.contains('Pedido #4821 confirmado').should('be.visible')
    cy.contains('En preparación').should('be.visible')
    cy.contains(`${harina.nombre} × 2`).should('be.visible')
    cy.get('.resumen-total').should('contain', pesos(harina.precio * 2))
    verCarrito(0)
  })

  it('HU-08: pago rechazado no registra el pedido y permite reintentar', () => {
    // Rechazado: 200 sin id. Cualquier otro resultado: pedido confirmado.
    cy.intercept({ method: 'POST', pathname: '/functions/v1/confirmar-pedido' }, (req) => {
      const rechazado = req.body.resultadoCobro === 'rechazado'
      responder(req, rechazado ? { estado: 'rechazado' } : pedidoOk(2), rechazado ? 200 : 201)
    }).as('confirmar')

    llegarAPago(2)
    completarPaso1Retiro()
    elegirResultado('Rechazado')
    cy.wait('@confirmar')

    // Sigue en /pago con el aviso y el carrito intacto
    cy.location('pathname').should('eq', '/pago')
    cy.get('[role="alert"]').should('contain', 'El cobro fue rechazado')
    verCarrito(2)
    cy.get('.resumen').should('contain', `${harina.nombre} × 2`)
    cy.get('.resumen-total').should('contain', pesos(harina.precio * 2))

    // Reintento con otro resultado
    elegirResultado('Aprobado')
    cy.wait('@confirmar').its('request.body.resultadoCobro').should('eq', 'aprobado')
    cy.location('pathname').should('eq', '/confirmacion')
    cy.contains('Pedido #4821 confirmado').should('be.visible')
  })

  it('HU-06: sin stock suficiente muestra el producto y deja el carrito intacto', () => {
    // La función responde 409 con el nombre del producto
    cy.intercept({ method: 'POST', pathname: '/functions/v1/confirmar-pedido' }, (req) =>
      responder(req, { error: 'Stock insuficiente', producto: harina.nombre }, 409)
    ).as('confirmar')

    llegarAPago(2)
    completarPaso1Retiro()
    elegirResultado('Aprobado')
    cy.wait('@confirmar')

    cy.get('[role="alert"]').should('have.text', `No alcanza el stock de ${harina.nombre}.`)
    cy.location('pathname').should('eq', '/pago')
    verCarrito(2)
    cy.get('.resumen').should('contain', `${harina.nombre} × 2`)
    cy.get('.resumen-total').should('contain', pesos(harina.precio * 2))
  })

  it('HU-07: retiro vs envío, costo por zona y zona sin cobertura', () => {
    const subtotal = harina.precio * 2
    llegarAPago(2)

    // Retiro: envío $ 0 y sin domicilio ni código postal
    cy.contains('label', 'Retiro en el local').find('input').check()
    cy.contains('label', `envío ${pesos(0)}`).should('be.visible')
    cy.get('#domicilio').should('not.exist')
    cy.get('#cp').should('not.exist')
    cy.get('.resumen-linea').should('contain', pesos(0))
    cy.get('.resumen-total').should('contain', pesos(subtotal))

    // Envío: aparecen los campos; CP 1754 cuesta 1500 y se suma al total
    cy.contains('label', 'Envío a domicilio').find('input').check()
    cy.get('#domicilio').should('be.visible')
    cy.get('#cp').type('1754')
    cy.contains('button', 'Verificar zona').click()
    cy.contains(`Costo de envío ${pesos(1500)}`).should('be.visible')
    cy.get('.resumen-linea').should('contain', pesos(1500))
    cy.get('.resumen-total').should('contain', pesos(subtotal + 1500))

    // Volver a retiro: el total vuelve al subtotal
    cy.contains('label', 'Retiro en el local').find('input').check()
    cy.get('.resumen-linea').should('contain', pesos(0))
    cy.get('.resumen-total').should('contain', pesos(subtotal))

    // CP sin zona: costo $ 0 y se puede avanzar al paso 2
    cy.contains('label', 'Envío a domicilio').find('input').check()
    cy.get('#cp').clear().type('9999')
    cy.contains('button', 'Verificar zona').click()
    cy.contains(`Costo de envío ${pesos(0)}`).should('be.visible')
    cy.get('#domicilio').type('Calle Falsa 123')
    cy.get('#nombre-pedido').type('{selectall}Ana Prueba')
    cy.get('#telefono').type('1100000000')
    cy.get('#correo-pedido').type(`{selectall}${USUARIO.email}`)
    cy.contains('button', 'Continuar').click()
    cy.contains('Cobro simulado').should('be.visible')
    cy.get('.resumen-total').should('contain', pesos(subtotal))
  })
})

import { bloquearSupabase, responder, usuarioAuth, verificarSinNoPrevistos } from './ayudas'

// Rellena el panel "Crear cuenta" con datos inventados (contraseña aleatoria)
function completarAlta() {

# Dietética Los Mirasoles

Proyecto final de Taller de Desarrollo de Aplicaciones.

Tienda web para un comercio de San Justo, Buenos Aires. Catálogo con filtros de aptitud, carrito, checkout con cobro simulado, retiro en el local y envío en GBA Oeste.

La tienda publicada está en https://tda-ecommerce-losmirasoles.onrender.com.

Las tres fases están en [`docs/`](docs/). Este archivo se va completando con lo que ya funciona.

## Stack

- React con Vite para las pantallas y el carrito en memoria.
- Supabase para Postgres, Auth, las imágenes y los datos de la tienda.
- La función `confirmar-pedido` graba el pedido, descuenta stock y manda el comprobante.
- El comprobante sale por SMTP de Gmail. La contraseña de aplicación no va en este repositorio.

## Cómo correrla

```bash
npm install
npm run dev
```

Hacen falta `VITE_SUPABASE_URL` y `VITE_SUPABASE_ANON_KEY`. En este repo ya están en `.env`.

## Qué hay hoy

La tienda recorre el circuito de compra y el panel.

- El catálogo es público: búsqueda, filtros de aptitud y ficha. Si el producto ya está en el carrito, el catálogo y la ficha muestran el mismo − / cantidad / +.
- El carrito vive en memoria. Sin sesión, "Seguir comprando" lleva al ingreso y después vuelve al catálogo. "Ir a pagar" aparece con sesión y con ítems.
- El checkout tiene dos pasos: retiro o envío, y cobro simulado (aprobado, rechazado o pendiente). Un código postal que no está en la tabla no frena el envío; el costo queda en cero.
- Confirmar llama a `confirmar-pedido`. Aprobado o pendiente graba el pedido y manda el comprobante al correo del checkout. Rechazado no graba nada, no manda correo y deja el carrito. Si el SMTP falla, el pedido queda igual y `mail_ok` queda en falso. La confirmación muestra ese pedido; recargar la pantalla no inventa un número.
- El panel está en `/admin/pedidos` y `/admin/productos`. Entra por el mismo ingreso. Guarda alta, edición, publicación, precio y stock, y cambia el estado del pedido. Al entrar en preparación, listo para retiro, despachado o entregado descuenta el stock una sola vez; al salir, lo devuelve. Sin sesión, esa ruta vuelve al ingreso. Un cliente autenticado ve el aviso y no entra.
- Las pantallas se acomodan en una columna. El circuito de compra entra en 360 px de ancho.

## Cuentas

Las dos usan `/ingreso`.

| Cuenta | Correo | Contraseña |
| --- | --- | --- |
| Cliente de prueba | `cliente.prueba@mail.com` | `Mirasoles.prueba` |
| Administrador | `administrador_losmirasoles@mail.com` | `Mirasoles.admin` |

El cliente ve la tienda. El administrador entra al panel y no ve el catálogo, el carrito ni el checkout.

## Supabase

Proyecto `Los Mirasoles (TDA)`. Las columnas están en el [diseño](<docs/U3 - TDA - Diseno.md>).

| Pieza | Qué guarda |
| --- | --- |
| `categorias` | Las siete líneas del catálogo, ya cargadas |
| `productos` | Precio, stock, publicado, aptitud (sin TACC, vegano, sin lactosa) e imagen |
| `zonas_cobertura` | Código postal y costo de envío. Hoy están 1706 y 1754. Si el código no está, el envío sigue y el costo queda en cero |
| `perfiles` | Una fila por usuario de Auth. El alta crea rol `cliente`. Hay una cuenta precargada con rol `administrador` |
| `pedidos` | Número visible, estado, cobro, entrega, contacto, `mail_ok` y si el stock ya fue descontado |
| `pedido_items` | Nombre, cantidad y precio copiados al confirmar |
| `pedido_cambios` | Fecha de cada estado. La escribe la base cuando el pedido se crea o cambia de estado |
| depósito `productos` | Una imagen de catálogo por producto. Lectura pública. Escribe el administrador |
| `confirmar-pedido` | Única escritura del alta de un pedido y del stock del checkout. Después manda el comprobante |

El navegador lee el catálogo. No inserta pedidos. La confirmación de correo de Auth está apagada, así que el registro deja la sesión abierta. El comprobante no sale por Auth: lo manda `confirmar-pedido` con `GMAIL_USER` y `GMAIL_APP_PASSWORD`, que viven en los secretos de la función y no en este repositorio.

import { createContext, useContext, useEffect, useRef, useState } from 'react'
import { supabase } from '../lib/supabase'

const AuthContext = createContext(null)

export function AuthProvider({ children }) {
  const [usuario, setUsuario] = useState(null)
  const [perfil, setPerfil] = useState(null)
  const [perfilListo, setPerfilListo] = useState(false)
  const [cargando, setCargando] = useState(true)
  const lectura = useRef(0)
  const usuarioVisto = useRef(undefined)

  async function leerPerfil(user) {
    const actual = ++lectura.current
    const id = user?.id ?? null
    const mismoUsuario = usuarioVisto.current === id
    usuarioVisto.current = id
    if (!user) {
      if (actual !== lectura.current) return
      setPerfil(null)
      setPerfilListo(true)
      return
    }
    // Only blank the role when the account changes. A token refresh
    // must not unmount the panel while the same profile is loading again.
    if (!mismoUsuario) setPerfilListo(false)
    const { data } = await supabase
      .from('perfiles')
      .select('nombre, rol')
      .eq('id', user.id)
      .maybeSingle()
    if (actual !== lectura.current) return
    setPerfil(data)
    setPerfilListo(true)
  }

  useEffect(() => {
    let vigente = true

    // Stay on "loading" until the role is known, so the panel does not
    // redirect an administrator before the profile arrives.
    supabase.auth.getSession().then(async ({ data }) => {
      const user = data.session?.user ?? null
      if (!vigente) return
      setUsuario(user)
      await leerPerfil(user)
      if (vigente) setCargando(false)
    })

    const { data } = supabase.auth.onAuthStateChange((_evento, sesion) => {
      const user = sesion?.user ?? null
      setUsuario(user)
      leerPerfil(user)
    })

    return () => {
      vigente = false
      data.subscription.unsubscribe()
    }
  }, [])

  // Cierra la sesión en Supabase; el "timbre" de arriba pone usuario en null
  async function cerrarSesion() {
    await supabase.auth.signOut()
  }

  return (
    <AuthContext.Provider value={{ usuario, perfil, perfilListo, cargando, cerrarSesion }}>
      {children}
    </AuthContext.Provider>
  )
}

export function useAuth() {
  return useContext(AuthContext)
}

import { createContext, useContext, useState} from 'react'

const CartContext = createContext(null)

export function CartProvider({ children }){
    
    const [items, setItems] = useState([])

    function agregarProducto(producto){
        const existe = items.find((i) => i.id === producto.id) 
        if (existe){
            // Math.min: la cantidad nunca pasa del stock
            setItems(items.map((i) => (i.id === producto.id ? { ...i, cantidad: Math.min(i.cantidad + 1, i.stock) } : i)))
        }else{
            setItems([...items, { ...producto, cantidad: 1 }])
        }
    }

    function quitarProducto(producto){
        setItems(items.filter((i) => i.id !== producto.id))
    }

    // cambio es +1 o -1; si la cantidad llega a 0, el producto sale del carrito
    // y nunca pasa del stock
    function cambiarCantidad(producto, cambio){
        setItems(
            items
                .map((i) => (i.id === producto.id ? { ...i, cantidad: Math.min(i.cantidad + cambio, i.stock) } : i))
                .filter((i) => i.cantidad > 0)
        )
    }

    function vaciarCarrito(){
        setItems([])
    }

    // Suma de precio por cantidad de todos los productos
    const subtotal = items.reduce((suma, i) => suma + i.precio * i.cantidad, 0)

    return (
    <CartContext.Provider value={{ items, subtotal, agregarProducto, quitarProducto, cambiarCantidad, vaciarCarrito }}>
      {children}
    </CartContext.Provider>
  )
}

export function useCart() {
  return useContext(CartContext) 
}

// Public URL for a product image stored in the productos bucket.
export function urlImagenProducto(path) {
  if (!path) return null
  if (path.startsWith('http://') || path.startsWith('https://')) return path
  const { data } = supabase.storage.from('productos').getPublicUrl(path)
  return data?.publicUrl ?? null
}

const CON_STOCK = new Set(['en_preparacion', 'listo_retiro', 'despachado', 'entregado'])

function etiquetaEstado(valor) {
  return ESTADOS.find(([clave]) => clave === valor)?.[1] ?? valor
}

function fechaLocal(valor) {
  if (!valor) return ''
  return new Date(valor).toLocaleString('es-AR', { dateStyle: 'short', timeStyle: 'short' })
}

function ultimaMarca(pedido) {
  const marcas = pedido.pedido_cambios ?? []
  return marcas.reduce((max, marca) => (marca.en > max ? marca.en : max), '')
}

// Retiro and shipping share one cycle step, with a different label.
function opcionesDe(pedido) {
  return ESTADOS.filter(([valor]) => {
    if (!valor) return false
    if (valor === 'listo_retiro') return pedido.modalidad === 'retiro' || pedido.estado === valor
    if (valor === 'despachado') return pedido.modalidad === 'envio' || pedido.estado === valor
    return true
  })
}

function textoDeCambio(anterior, siguiente) {
  const tenia = CON_STOCK.has(anterior)
  const pide = CON_STOCK.has(siguiente)
  if (!tenia && pide) return 'Estado actualizado. Se descontó el stock.'
  if (tenia && !pide) return 'Estado actualizado. El stock volvió al catálogo.'
  return 'Estado actualizado.'
}

function PedidosAdmin() {
  const [pedidos, setPedidos] = useState([])
  const [estado, setEstado] = useState('')
  const [entrega, setEntrega] = useState('')
  const [desde, setDesde] = useState('')
  const [hasta, setHasta] = useState('')
  const [buscar, setBuscar] = useState('')
  const [mensaje, setMensaje] = useState('')
  const [guardando, setGuardando] = useState(null)

  useEffect(() => {
    cargar()
  }, [])

  async function cargar() {
    const { data, error } = await supabase
      .from('pedidos')
      .select('*, pedido_cambios(en)')
      .order('id', { ascending: false })
    if (error) {
      setMensaje('No se pudieron cargar los pedidos.')
      return
    }
    setPedidos(data ?? [])
  }

  async function cambiarEstado(pedido, siguiente) {
    if (siguiente === pedido.estado) return
    setGuardando(pedido.id)
    setMensaje('')
    const { error } = await supabase.from('pedidos').update({ estado: siguiente }).eq('id', pedido.id)
    setGuardando(null)
    if (error) {
      const texto = error.message || ''
      if (error.code === 'P0001' && texto) {
        setMensaje(texto.endsWith('.') ? texto : `${texto}.`)

// Shop-counter wording. The database errors stay off this screen.
function validarProducto(formulario) {
  const errores = {}
  if (!formulario.nombre.trim()) errores.nombre = 'Indicá el nombre.'

  const precioTexto = String(formulario.precio).trim().replace(',', '.')
  const precio = Number(precioTexto)
  if (!precioTexto || Number.isNaN(precio)) errores.precio = 'Indicá el precio.'
  else if (precio <= 0) errores.precio = 'El precio tiene que ser mayor a cero.'

  const stockTexto = String(formulario.stock).trim()
  const stock = Number(stockTexto)
  if (!stockTexto || !Number.isInteger(stock) || stock < 0) {
    errores.stock = 'Indicá el stock en unidades enteras, desde cero.'
  }

  if (!formulario.categoria_id) errores.categoria_id = 'Elegí una categoría.'

  return { errores, precio, stock }
}

function mensajeDeBase(error, fallback) {
  if (!error) return ''
  if (error.code === '23503') return 'No se puede eliminar: figura en un pedido.'
  return fallback
}

// Pantalla de inicio: trae los productos publicados de Supabase
function Catalogo() {
  const { items } = useCart()
  const [productos, setProductos] = useState([])
  const [cargando, setCargando] = useState(true)
  const [mensajeError, setMensajeError] = useState('')
  const [busqueda, setBusqueda] = useState('')
  const [categorias, setCategorias] = useState([])
  const [categoriaElegida, setCategoriaElegida] = useState('')
  const [aptitudes, setAptitudes] = useState({
    sin_tacc: false,
    vegano: false,
    sin_lactosa: false,
  })

  // Una sola vez al abrir la pantalla: pide los productos y las categorías
  useEffect(() => {
    async function traerDatos() {
      const resProductos = await supabase
        .from('productos')
        .select('*')
        .eq('publicado', true)
      const resCategorias = await supabase
        .from('categorias')
        .select('*')
        .order('nombre')

      if (resProductos.error || resCategorias.error) {
        setMensajeError('No se pudieron cargar los productos')
      } else {
        setProductos(resProductos.data)
        setCategorias(resCategorias.data)
      }
      setCargando(false)
    }

    traerDatos()
  }, [])

  if (cargando) {
    return <main className="pagina"><p>Cargando...</p></main>
  }

  if (mensajeError) {
    return <main className="pagina"><p role="alert">{mensajeError}</p></main>
  }

  // Cuántas unidades de ese producto ya hay en el carrito
  function cantidadEnCarrito(id) {
    const item = items.find((i) => i.id === id)
    return item ? item.cantidad : 0
  }

  // Checked dietary flags all have to match. None checked means no extra filter.
  const productosFiltrados = productos.filter(
    (p) =>
      normalizar(p.nombre).includes(normalizar(busqueda)) &&
      (categoriaElegida === '' || p.categoria_id === categoriaElegida) &&
      (!aptitudes.sin_tacc || p.sin_tacc) &&
      (!aptitudes.vegano || p.vegano) &&
      (!aptitudes.sin_lactosa || p.sin_lactosa)
  )

  return (
    <main className="pagina catalogo">
      <p className="ceja">Catálogo</p>
      <div className="catalogo-tope">
        <div className="chips">
          <button
            type="button"
            className="chip"
            aria-pressed={categoriaElegida === ''}
            onClick={() => setCategoriaElegida('')}
          >
            Todas
          </button>
          {categorias.map((categoria) => (
            <button
              key={categoria.id}
              type="button"
              className="chip"
              aria-pressed={categoriaElegida === categoria.id}
              onClick={() => setCategoriaElegida(categoria.id)}
            >
              {categoria.nombre}
            </button>
          ))}
        </div>
        <label className="buscar">
          <Icono nombre="buscar" />
          <input
            className="campo campo-buscar"
            type="search"
            placeholder="Buscar producto"
            aria-label="Buscar producto"
            value={busqueda}
            onChange={(e) => setBusqueda(e.target.value)}
          />
        </label>
      </div>

      <div className="filtros">
        {APTITUDES.map((aptitud) => (
          <label key={aptitud.clave} className="check">
            <input
              type="checkbox"
              checked={aptitudes[aptitud.clave]}
              onChange={() =>
                setAptitudes((prev) => ({ ...prev, [aptitud.clave]: !prev[aptitud.clave] }))
              }
            />
            {aptitud.etiqueta}
          </label>
        ))}
      </div>

      {productosFiltrados.length === 0 && (
        <p>No hay productos que coincidan con tu búsqueda.</p>
      )}

      <ul className="grilla">
        {productosFiltrados.map((producto) => {
          const etiquetas = etiquetasDe(producto)
          const sinStock = producto.stock <= 0
          return (
            <li key={producto.id} className="tarjeta">
              <Link to={`/producto/${producto.id}`} aria-label={producto.nombre}>
                <MarcoImagen src={urlImagenProducto(producto.imagen_path)} alt="" />
              </Link>
              <h2 className="tarjeta-nombre">
                <Link to={`/producto/${producto.id}`}>{producto.nombre}</Link>
              </h2>
              {etiquetas.length > 0 && (
                <p className="chips chips-chicos">
                  {etiquetas.map((etiqueta) => (
                    <span key={etiqueta} className="chip chip-fijo">{etiqueta}</span>
                  ))}
                </p>
              )}
              <p className="precio">{pesos(producto.precio)}</p>
              {sinStock ? (
                <span className="chip chip-aviso">Sin stock</span>
              ) : (
                <ControlCantidad producto={producto} cantidad={cantidadEnCarrito(producto.id)} />
              )}
            </li>
          )
        })}
      </ul>
    </main>
  )
}

function mensajeDeAlta(error, data) {
  const correoRepetido = /already registered/i.test(error?.message ?? '')
    || error?.code === 'user_already_exists'
  const altaDeshabilitada = error?.code === 'email_provider_disabled'
    || /signups are disabled/i.test(error?.message ?? '')
  const yaExiste = correoRepetido || (!error && data?.user && (data.user.identities?.length ?? 0) === 0)
  if (!error && !yaExiste) return ''
  if (yaExiste) return 'Ese correo ya está registrado.'
  if (altaDeshabilitada) return 'El alta por correo está deshabilitada en el proyecto.'
  return 'No se pudo crear la cuenta'
}

// Only in-app paths. Anything else falls back to the catalog.
function destinoDe(desde) {
  if (typeof desde !== 'string' || !desde.startsWith('/') || desde.startsWith('//')) return '/'
  return desde
}

function Ingreso() {
  const navegar = useNavigate()
  const destino = destinoDe(useLocation().state?.desde)
  const [nombre, setNombre] = useState('')
  const [correoAlta, setCorreoAlta] = useState('')
  const [contraseniaAlta, setContraseniaAlta] = useState('')
  const [correo, setCorreo] = useState('')
  const [contrasenia, setContrasenia] = useState('')
  const [errorAlta, setErrorAlta] = useState('')
  const [errorIngreso, setErrorIngreso] = useState('')

  async function entrar(e) {
    e.preventDefault()
    const { data, error } = await supabase.auth.signInWithPassword({
      email: correo,
      password: contrasenia,
    })
    setErrorIngreso(error ? 'Correo o contraseña incorrectos' : '')
    if (error) return
    const { data: fila } = await supabase
      .from('perfiles')
      .select('rol')
      .eq('id', data.user.id)
      .maybeSingle()
    // The administrator stays in the panel. The store is the customer's side.
    if (fila?.rol === 'administrador') {
      const panel = destino.startsWith('/admin') ? destino : '/admin/pedidos'
      navegar(panel, { replace: true })
      return
    }
    navegar(destino)
  }

  async function crearCuenta(e) {
    e.preventDefault()
    const { data, error } = await supabase.auth.signUp({
      email: correoAlta,
      password: contraseniaAlta,
      options: { data: { nombre: nombre.trim() } },
    })
    const mensaje = mensajeDeAlta(error, data)
    setErrorAlta(mensaje)
    if (!mensaje) navegar(destino)
  }

  return (

      <p className="ceja">Cuenta</p>

            id="correo"

            onChange={(e) => setCorreo(e.target.value)}
            required
          />
          <label className="etiqueta" htmlFor="contrasenia">Contraseña</label>
          <input
            className="campo"
            id="contrasenia"
            type="password"
            placeholder="••••••••"
            value={contrasenia}
            onChange={(e) => setContrasenia(e.target.value)}
            required
          />
          {errorIngreso && <p role="alert">{errorIngreso}</p>}
          <button className="boton" type="submit">Entrar</button>
          <p className="nota">¿No tenés cuenta? Usá el panel de al lado.</p>
        </form>

        <form className="panel" onSubmit={crearCuenta}>

// Same lookup the checkout function uses. An unknown postal code stays at zero.
async function costoDeZona(codigo) {
  const { data, error } = await supabase
    .from('zonas_cobertura')
    .select('costo_envio')
    .eq('codigo_postal', codigo.trim())
    .maybeSingle()
  if (error) return null
  return data ? Number(data.costo_envio) : 0
}

// Reads the function error body. Stock conflicts name the product; other failures use the server text.
async function mensajeDeConfirmacion(error) {
  const respuesta = error?.context
  if (!respuesta || typeof respuesta.json !== 'function') {
    return 'No se pudo confirmar el pedido.'
  }
  try {
    const cuerpo = await respuesta.json()
    if (cuerpo?.producto) return `No alcanza el stock de ${cuerpo.producto}.`
    if (typeof cuerpo?.error === 'string' && cuerpo.error) return cuerpo.error
  } catch {
    // The function did not return JSON.
  }
  return 'No se pudo confirmar el pedido.'
}

// Two checkout steps. Confirm sends the cart to confirmar-pedido; prices and stock are decided there.
function Pago() {
  const { items, subtotal, vaciarCarrito } = useCart()
  const { usuario, perfil } = useAuth()
  const navegar = useNavigate()
  const [paso, setPaso] = useState(1)
  const [modalidad, setModalidad] = useState('envio')
  const [domicilio, setDomicilio] = useState('')
  const [codigoPostal, setCodigoPostal] = useState('')
  const [nombre, setNombre] = useState(perfil?.nombre ?? '')
  const [telefono, setTelefono] = useState('')
  const [correo, setCorreo] = useState(usuario?.email ?? '')
  const [costoEnvio, setCostoEnvio] = useState(0)
  const [zonaConsultada, setZonaConsultada] = useState(false)
  const [resultado, setResultado] = useState('aprobado')
  const [enviando, setEnviando] = useState(false)
  const [mensaje, setMensaje] = useState('')

  const envio = modalidad === 'envio' ? costoEnvio : 0

  // Auth can arrive after the first render. Fill only fields the user has not typed.
  useEffect(() => {
    if (!nombre && perfil?.nombre) setNombre(perfil.nombre)
    if (!correo && usuario?.email) setCorreo(usuario.email)
  }, [nombre, correo, perfil, usuario])

  async function confirmarPedido(evento) {
    evento.preventDefault()
    if (!usuario || items.length === 0 || enviando) return
    setEnviando(true)
    setMensaje('')

    const { data, error } = await supabase.functions.invoke('confirmar-pedido', {
      body: {
        items: items.map((item) => ({ productoId: item.id, cantidad: item.cantidad })),
        entrega: {
          modalidad,
          domicilio: modalidad === 'envio' ? domicilio.trim() : null,
          codigoPostal: modalidad === 'envio' ? codigoPostal.trim() : null,
        },
        contacto: {
          nombre: nombre.trim(),
          telefono: telefono.trim(),
          correo: correo.trim(),
        },
        resultadoCobro: resultado,
      },
    })

    setEnviando(false)
    if (error) {
      setMensaje(await mensajeDeConfirmacion(error))
      return
    }
    // A rejected charge is a 200 with no order. The cart stays as it is.
    if (!data?.id) {
      setMensaje('El cobro fue rechazado. El pedido no se registró.')
      return
    }
    vaciarCarrito()
    navegar('/confirmacion', { state: data })
  }

  async function verificarZona() {
    const costo = await costoDeZona(codigoPostal)
    if (costo === null) return
    setCostoEnvio(costo)
    setZonaConsultada(true)
  }

  // Refresh the zone before step 2 so the summary matches the total that will be stored.
  async function continuar(evento) {
    evento.preventDefault()
    if (modalidad === 'envio') {
      const costo = await costoDeZona(codigoPostal)
      if (costo !== null) {
        setCostoEnvio(costo)
        setZonaConsultada(true)
      }
    }
    setPaso(2)
  }

  return (
    <main className="pagina">
      <p className="ceja">
        Pago · paso {paso} de 2 · {paso === 1 ? 'Entrega' : 'Cobro simulado'}
      </p>
      <div className="pago">
        <section>
          {paso === 1 ? (
            <>
              <h1 className="titulo">Entrega y datos</h1>
              <form className="formulario" onSubmit={continuar}>
                <label className="opcion">
                  <input
                    type="radio"
                    name="modalidad"
                    checked={modalidad === 'retiro'}
                    onChange={() => setModalidad('retiro')}
                  />
                  <Icono nombre="local" />
                  Retiro en el local · envío {pesos(0)}
                </label>
                <label className="opcion">
                  <input
                    type="radio"
                    name="modalidad"
                    checked={modalidad === 'envio'}
                    onChange={() => setModalidad('envio')}
                  />
                  <Icono nombre="envio" />
                  Envío a domicilio
                </label>

                {modalidad === 'envio' && (
                  <>
                    <label className="etiqueta" htmlFor="domicilio">Calle, altura y localidad</label>
                    <input
                      className="campo"
                      id="domicilio"
                      value={domicilio}
                      placeholder="San Justo"
                      onChange={(e) => setDomicilio(e.target.value)}
                      required
                    />
                    <label className="etiqueta" htmlFor="cp">Código postal</label>
                    <div className="linea-cp">
                      <input
                        className="campo"
                        id="cp"
                        value={codigoPostal}
                        placeholder="1754"
                        onChange={(e) => {
                          setCodigoPostal(e.target.value)
                          setZonaConsultada(false)
                          setCostoEnvio(0)
                        }}
                        required
                      />
                      <button className="boton" type="button" onClick={verificarZona}>
                        Verificar zona
                      </button>
                      {zonaConsultada && <span>Costo de envío {pesos(costoEnvio)}</span>}
                    </div>
                  </>
                )}

                <div className="dos">
                  <div>
                    <label className="etiqueta" htmlFor="nombre-pedido">Nombre</label>
                    <input
                      className="campo"
                      id="nombre-pedido"
                      value={nombre}
                      placeholder="Nombre y apellido"
                      onChange={(e) => setNombre(e.target.value)}
                      required
                    />
                  </div>
                  <div>
                    <label className="etiqueta" htmlFor="telefono">Teléfono</label>
                    <input

function receipt(order: Order): string {

class Smtp {
  #conn: Deno.Conn;
  #reader: ReadableStreamDefaultReader<Uint8Array>;
  #writer: WritableStreamDefaultWriter<Uint8Array>;
  #pending = "";

  constructor(conn: Deno.Conn) {
    this.#conn = conn;
    this.#reader = conn.readable.getReader();
    this.#writer = conn.writable.getWriter();
  }

  async #line(): Promise<string> {
    const decoder = new TextDecoder();
    while (!this.#pending.includes("\n")) {
      const { value, done } = await this.#reader.read();
      if (done) throw new Error("SMTP connection closed");
      this.#pending += decoder.decode(value, { stream: true });
    }
    const end = this.#pending.indexOf("\n");
    const line = this.#pending.slice(0, end).replace(/\r$/, "");
    this.#pending = this.#pending.slice(end + 1);
    return line;
  }

  async reply(): Promise<number> {
    let line = await this.#line();
    const code = Number(line.slice(0, 3));
    while (line.length >= 4 && line[3] === "-") {
      line = await this.#line();
    }
    if (!Number.isInteger(code)) throw new Error("Invalid SMTP reply");
    return code;
  }

  async command(value: string): Promise<number> {
    await this.#writer.write(new TextEncoder().encode(value));
    return await this.reply();
  }

  // STARTTLS on port 587. Locks are released so the handshake can own the socket.
  async startTls(): Promise<void> {
    this.#reader.releaseLock();
    this.#writer.releaseLock();
    const tls = await Deno.startTls(this.#conn, { hostname: SMTP_HOST });
    this.#conn = tls;
    this.#reader = tls.readable.getReader();
    this.#writer = tls.writable.getWriter();
    this.#pending = "";
  }

  close(): void {
    try {
      this.#conn.close();
    } catch {
      // The timeout path may already have closed the socket.
    }
  }
}

function dotStuff(body: string): string {
  return body
    .replace(/\r?\n/g, "\r\n")
    .split("\r\n")
    .map((line) => (line.startsWith(".") ? `.${line}` : line))
    .join("\r\n");
}

async function deliver(smtp: Smtp, from: string, to: string, subject: string, body: string): Promise<void> {
  if (await smtp.reply() !== 220) throw new Error("SMTP greeting rejected");
  if (await smtp.command("EHLO losmirasoles\r\n") !== 250) throw new Error("SMTP EHLO rejected");
  if (await smtp.command("STARTTLS\r\n") !== 220) throw new Error("SMTP STARTTLS rejected");
  await smtp.startTls();
  if (await smtp.command("EHLO losmirasoles\r\n") !== 250) throw new Error("SMTP EHLO after TLS rejected");
  if (await smtp.command("AUTH LOGIN\r\n") !== 334) throw new Error("SMTP AUTH rejected");
  if (await smtp.command(`${btoa(from)}\r\n`) !== 334) throw new Error("SMTP user rejected");
  const password = (Deno.env.get("GMAIL_APP_PASSWORD") ?? "").replace(/\s+/g, "");
  if (await smtp.command(`${btoa(password)}\r\n`) !== 235) throw new Error("SMTP password rejected");
  if (await smtp.command(`MAIL FROM:<${from}>\r\n`) !== 250) throw new Error("SMTP MAIL FROM rejected");
  if (await smtp.command(`RCPT TO:<${to}>\r\n`) !== 250) throw new Error("SMTP RCPT TO rejected");
  if (await smtp.command("DATA\r\n") !== 354) throw new Error("SMTP DATA rejected");

  const message = [
    `From: ${from}`,
    `To: ${to}`,
    `Subject: ${subject}`,
    "MIME-Version: 1.0",
    "Content-Type: text/plain; charset=UTF-8",
    "",
    dotStuff(body),
    "",
  ].join("\r\n");
  if (await smtp.command(`${message}\r\n.\r\n`) !== 250) throw new Error("SMTP message rejected");
  await smtp.command("QUIT\r\n").catch(() => undefined);
}

async function sendReceipt(order: Order, to: string): Promise<void> {
  const from = (Deno.env.get("GMAIL_USER") ?? "").trim();
  const conn = await Deno.connect({ hostname: SMTP_HOST, port: SMTP_PORT });
  const smtp = new Smtp(conn);
  let timer: number | undefined;
  try {
    await Promise.race([
      deliver(smtp, from, to, `Pedido ${order.id}`, receipt(order)),
      new Promise<never>((_, reject) => {
        timer = setTimeout(() => {
          smtp.close();
          reject(new Error("SMTP timeout"));
        }, SMTP_TIMEOUT_MS);
      }),
    ]);
  } finally {
    if (timer !== undefined) clearTimeout(timer);
    smtp.close();
  }
}

Deno.serve(async (req: Request) => {
  if (req.method === "OPTIONS") {
    return new Response(null, { status: 204, headers: CORS });
  }
  if (req.method !== "POST") {
    return json(405, { error: "Método no permitido" });
  }

  const authHeader = bearer(req.headers.get("Authorization"));
  if (!authHeader) return json(401, { error: "Se requiere sesión" });

  const perfilId = await userId(authHeader);
  if (!perfilId) return json(401, { error: "Se requiere sesión" });

  let body: Record<string, unknown>;
  try {
    body = await req.json();
  } catch {
    return json(422, { error: "Falta un dato obligatorio" });
  }

  const entrega = (body.entrega ?? {}) as Record<string, unknown>;
  const contacto = (body.contacto ?? {}) as Record<string, unknown>;
  const items = itemsForRpc(body.items);
  if (!items) return json(422, { error: "Falta un dato obligatorio" });

  let result: CheckoutResult;
  try {
    result = await callCheckout({
      p_perfil_id: perfilId,
      p_items: items,
      p_modalidad: asText(entrega.modalidad),
      p_domicilio: asText(entrega.domicilio),
      p_codigo_postal: asText(entrega.codigoPostal),
      p_nombre: asText(contacto.nombre),
      p_telefono: asText(contacto.telefono),
      p_correo: asText(contacto.correo),
      p_resultado: asText(body.resultadoCobro),
    });
  } catch {
    return json(500, { error: "No se pudo confirmar el pedido" });
  }

  if (result.status === 422) {
    return json(422, { error: "Falta un dato obligatorio" });
  }
  if (result.status === 409) {
    return json(409, { producto: result.producto });
  }
  if (result.status === 200) {
    return json(200, { pedido: null });
  }
  if (result.status !== 201 || !result.pedido) {
    return json(500, { error: "No se pudo confirmar el pedido" });
  }

  const order = result.pedido;
  if (!order.direccionLocal && order.modalidad === "retiro") {
    order.direccionLocal = LOCAL_PICKUP;
  }

  let mailOk = false;
  const gmailUser = (Deno.env.get("GMAIL_USER") ?? "").trim();
  const gmailPassword = (Deno.env.get("GMAIL_APP_PASSWORD") ?? "").trim();
  if (gmailUser && gmailPassword && contacto.correo) {
    try {
      await sendReceipt(order, String(contacto.correo).trim());
      mailOk = await markMailSent(order.id);
    } catch (error) {
      console.error("receipt mail failed", error instanceof Error ? error.message : "unknown");
      mailOk = false;
    }
  }

  return json(201, { ...order, mail_ok: mailOk });
});

create or replace function public.aplicar_cambio_estado_pedido()
returns trigger
language plpgsql
security definer
set search_path to public
as $$
declare
  v_faltante text;
  v_debe_tener boolean;
begin
  -- Ignore a flag sent by the browser. The rule below owns this column.
  new.stock_descontado := old.stock_descontado;

  if new.estado is not distinct from old.estado then
    return new;
  end if;

  if new.estado not in (
    'pendiente_confirmacion', 'recibido', 'en_preparacion',
    'listo_retiro', 'despachado', 'entregado', 'cancelado'
  ) then
    raise exception 'Ese estado no está permitido';
  end if;

  if new.estado = 'listo_retiro' and new.modalidad is distinct from 'retiro' then
    raise exception 'Este pedido es un envío: marcá despachado';
  end if;

  if new.estado = 'despachado' and new.modalidad is distinct from 'envio' then
    raise exception 'Este pedido es un retiro: marcá listo para retiro';
  end if;

  v_debe_tener := new.estado in (
    'en_preparacion', 'listo_retiro', 'despachado', 'entregado'
  );

  if v_debe_tener and not old.stock_descontado then
    perform 1
    from public.productos p
    where p.id in (
      select i.producto_id from public.pedido_items i where i.pedido_id = new.id
    )
    order by p.id
    for update;

    select p.nombre into v_faltante
    from (
      select producto_id, sum(cantidad)::integer as cantidad
      from public.pedido_items
      where pedido_id = new.id
      group by producto_id
    ) i
    join public.productos p on p.id = i.producto_id
    where p.stock < i.cantidad
    order by p.nombre
    limit 1;

    if v_faltante is not null then
      raise exception 'No alcanza el stock de %', v_faltante;
    end if;

    update public.productos p
    set stock = p.stock - i.cantidad
    from (
      select producto_id, sum(cantidad)::integer as cantidad
      from public.pedido_items
      where pedido_id = new.id
      group by producto_id
    ) i
    where p.id = i.producto_id;

    new.stock_descontado := true;
  elsif not v_debe_tener and old.stock_descontado then
    perform 1
    from public.productos p
    where p.id in (
      select i.producto_id from public.pedido_items i where i.pedido_id = new.id
    )
    order by p.id
    for update;

    update public.productos p
    set stock = p.stock + i.cantidad
    from (
      select producto_id, sum(cantidad)::integer as cantidad
      from public.pedido_items
      where pedido_id = new.id
      group by producto_id
    ) i
    where p.id = i.producto_id;

    new.stock_descontado := false;
  end if;

  insert into public.pedido_cambios (pedido_id, estado)
  values (new.id, new.estado);

  return new;
end;

  end if;

  for v_item in select value from jsonb_array_elements(p_items)
  loop
    if jsonb_typeof(v_item) is distinct from 'object'
       or jsonb_typeof(v_item->'productoId') is distinct from 'string'
       or jsonb_typeof(v_item->'cantidad') is distinct from 'number' then
      return jsonb_build_object('status', 422, 'error', 'Falta un dato obligatorio');
    end if;

    begin
      v_producto_id := (v_item->>'productoId')::uuid;
      v_cantidad := (v_item->>'cantidad')::integer;
    exception
      when invalid_text_representation or numeric_value_out_of_range then
        return jsonb_build_object('status', 422, 'error', 'Falta un dato obligatorio');
    end;

    if v_producto_id is null or v_cantidad is null or v_cantidad <= 0 then
      return jsonb_build_object('status', 422, 'error', 'Falta un dato obligatorio');
    end if;

    v_pos := array_position(v_ids, v_producto_id);
    if v_pos is null then
      v_ids := array_append(v_ids, v_producto_id);
      v_qtys := array_append(v_qtys, v_cantidad);
    else
      v_qtys[v_pos] := v_qtys[v_pos] + v_cantidad;
    end if;
  end loop;

  -- Lock in a stable order so two checkouts cannot both take the last unit.
  perform 1
  from public.productos
  where id = any (v_ids)
  order by id
  for update;

  if (
    select count(*)
    from public.productos
    where id = any (v_ids)
      and publicado
  ) <> coalesce(array_length(v_ids, 1), 0) then
    return jsonb_build_object('status', 422, 'error', 'Falta un dato obligatorio');
  end if;

  select p.nombre into v_nombre
  from unnest(v_ids, v_qtys) as t(producto_id, cantidad)
  join public.productos p on p.id = t.producto_id
  where p.stock < t.cantidad
  order by p.nombre
  limit 1;

  if found then
    return jsonb_build_object('status', 409, 'producto', v_nombre);
  end if;

  -- Rejected payment creates no row and sends no mail.
  if p_resultado = 'rechazado' then
    return jsonb_build_object('status', 200);
  end if;

  if p_modalidad = 'envio' then
    select z.costo_envio into v_costo
    from public.zonas_cobertura z
    where z.codigo_postal = btrim(p_codigo_postal);
    if not found then
      v_costo := 0;
    end if;
  end if;

  select coalesce(sum(p.precio * t.cantidad), 0) into v_subtotal
  from unnest(v_ids, v_qtys) as t(producto_id, cantidad)
  join public.productos p on p.id = t.producto_id;

  v_total := v_subtotal + v_costo;
  v_estado := case
    when p_resultado = 'aprobado' then 'en_preparacion'
    else 'pendiente_confirmacion'
  end;

  insert into public.pedidos (
    perfil_id, estado, resultado_cobro, modalidad, costo_envio, total,
    codigo_postal, domicilio, nombre, telefono, correo
  ) values (
    p_perfil_id, v_estado, p_resultado, p_modalidad, v_costo, v_total,
    case when p_modalidad = 'envio' then btrim(p_codigo_postal) else null end,
    case when p_modalidad = 'envio' then btrim(p_domicilio) else null end,
    btrim(p_nombre), btrim(p_telefono), btrim(p_correo)
  )
  returning id into v_pedido_id;

  insert into public.pedido_items (pedido_id, producto_id, nombre, cantidad, precio_unitario)
  select v_pedido_id, p.id, p.nombre, t.cantidad, p.precio
  from unnest(v_ids, v_qtys) as t(producto_id, cantidad)
  join public.productos p on p.id = t.producto_id;

  if p_resultado = 'aprobado' then
    update public.productos p
    set stock = p.stock - t.cantidad
    from unnest(v_ids, v_qtys) as t(producto_id, cantidad)
    where p.id = t.producto_id;
  end if;

  select coalesce(jsonb_agg(
    jsonb_build_object(
      'productoId', pi.producto_id,
      'nombre', pi.nombre,
      'cantidad', pi.cantidad,
      'precioUnitario', pi.precio_unitario
    )
    order by pi.nombre
  ), '[]'::jsonb)
  into v_items
  from public.pedido_items pi
  where pi.pedido_id = v_pedido_id;

  return jsonb_strip_nulls(jsonb_build_object(
    'status', 201,
    'pedido', jsonb_build_object(
      'id', v_pedido_id,
      'estado', v_estado,
      'items', v_items,
      'total', v_total,
      'modalidad', p_modalidad,
      'costoEnvio', v_costo,
      'direccionLocal', case
        when p_modalidad = 'retiro'
          then 'Dietética Los Mirasoles, San Justo, La Matanza'
        else null
      end
    )
  ));
end;
$$;
